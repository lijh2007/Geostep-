cmake_minimum_required(VERSION 3.28)
project(mockgeo_root LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(LSPLANT_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(DOBBY_GENERATE_SHARED OFF CACHE BOOL "" FORCE)
set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/lsplant)
add_subdirectory(vendor/dobby)
file(GLOB XDL_SOURCES vendor/xdl/*.c)
add_library(mockgeo_xdl STATIC ${XDL_SOURCES})
target_include_directories(mockgeo_xdl PUBLIC vendor/xdl/include vendor/xdl)
add_library(mockgeo_root SHARED bootstrap.cpp)
target_include_directories(mockgeo_root PRIVATE vendor/dobby/include)
target_link_libraries(mockgeo_root PRIVATE lsplant_static dobby mockgeo_xdl log dl)
target_link_options(mockgeo_root PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
set_target_properties(mockgeo_root PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/jni/arm64-v8a")
add_executable(mockgeo_loader injector.cpp)
target_link_libraries(mockgeo_loader PRIVATE dl)
target_link_options(mockgeo_loader PRIVATE "-Wl,-z,max-page-size=16384")
set_target_properties(mockgeo_loader PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tools")
